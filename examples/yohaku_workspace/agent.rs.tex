    let mut result = String::new();
    let mut rest = prompt;

        rest = &rest[start..];
        let Some(end) = rest.find('}') else {
            break;

            }
            "selection" => Some(""),

            _ => None,
        };

        rest = &rest[end + 1..];
    }

/// The connector reports either `time budget: <remaining>s of <total>s ...`
/// (finite) or `time budget: unlimited; elapsed <n>s ...` (no whole-turn limit).
/// For finite reports the first integer is the time still available at the moment
/// of the report and the second is the configured total; trailing prose such as
/// "remain; elapsed 5s; ..." never enters either number. Unlimited is kept as its
/// own state so the UI never invents a zero-second countdown. Malformed,
/// placeholder or absurd reports return None and keep the plain status.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub(crate) enum TurnBudget {
    Limited { remaining: u64, total: u64 },
    Unlimited,
}
pub(crate) fn parse_turn_budget(text: &str) -> Option<TurnBudget> {
    const MARKER: &str = "time budget:";
    let rest = text.split_once(MARKER)?.1.trim_start();
    if let Some(tail) = rest.strip_prefix("unlimited") {
        if tail.is_empty() || tail.starts_with(';') {
            return Some(TurnBudget::Unlimited);
        }
    }
    let (remaining_len, remaining) = leading_seconds(rest)?;
    let rest = rest.get(remaining_len..)?.strip_prefix('s')?.trim_start();
    let rest = rest.strip_prefix("of ")?.trim_start();
    let (_, total) = leading_seconds(rest)?;
    if !(1..=604_800).contains(&total) || remaining == 0 || remaining > total {
        return None;
    }
    Some(TurnBudget::Limited { remaining, total })
}
fn leading_seconds(text: &str) -> Option<(usize, u64)> {
    let end = text.find(|c: char| !c.is_ascii_digit())?;
    if end == 0 {
        return None;
    }
    let digits = text.get(..end)?;
    if !text.get(end..)?.starts_with('s') {
        return None;
    }
    Some((end, digits.parse::<u64>().ok()?))
}
/// Mirrors the connector's reported budget into the status line.
/// For a finite budget `since` is when that report arrived, so the first rendered
/// value equals its reported remaining time and later frames count down from there.
/// The elapsed time already included in the report is never subtracted twice.
/// For an unlimited turn `since` is the start of the turn, so the UI shows how long
/// the turn has actually been running and never invents a zero-second countdown.
/// It never stops or restarts a turn; without a report, or once the turn is over,
/// the status is shown unchanged.
pub(crate) fn budget_status(status: String, budget: Option<TurnBudget>, since: std::time::Instant, running: bool) -> String {
    // The connector's initial report is a snapshot. Keep one live clock rather
    // than showing its stale elapsed=0 next to the current elapsed value.
    let status = if running && budget.is_some() {
        status.split_once(" | time budget:").map(|(phase, _)| phase.to_owned()).unwrap_or(status)
    } else { status };
    match budget {
        Some(TurnBudget::Limited { remaining, total }) if running => {
            let since_report = since.elapsed().as_secs();
            let remaining = remaining.saturating_sub(since_report);
            let used = total.saturating_sub(remaining);
            format!("budget {remaining}s left of {total}s ({used}s used) | {status}")
        }
        Some(TurnBudget::Unlimited) if running => {
            format!("budget unlimited; {}s elapsed | {status}", since.elapsed().as_secs())
        }
        _ => status,
    }
}

    fn launch_worker(&mut self, mut command: Command, initial: Value, panel: &mut Panel) -> bool {
        let turn = self.begin_turn();
        let launch = command
            .stdin(Stdio::piped())
            .stdout(Stdio::piped())
            .stderr(Stdio::null())
            .spawn();
        let mut child = match launch {
            Ok(c) => c,
            Err(_) => {
                self.status = "Could not start Python 3; check that it is installed".into();
                return false;
            }
        };
        let mut stdin = child.stdin.take().unwrap();
        let stdout = child.stdout.take().unwrap();
        if writeln!(stdin, "{initial}").is_err() {
            let _ = child.kill();
            let _ = child.wait();
            self.status = "Could not send request to AI connector".into();
            return false;
        }
        let child = Arc::new(Mutex::new(child));
        self.child = Some(child.clone());
        let tx = self.tx.clone();
        std::thread::spawn(move || {
            for line in BufReader::new(stdout).lines() {
                let value = match line
                    .ok()
                    .and_then(|s| serde_json::from_str::<Value>(&s).ok())
                {
                    Some(v) => v,
                    None => break,
                };
                if value["event"] == "request" {
                    let local_id = value["id"].as_u64().unwrap_or(0);
                    let id = (turn << 32) | local_id;
                    let (reply, rx) = mpsc::channel();
                    if tx
                        .send(Event::Request(
                            id,
                            {
                                let mut request = value["request"].clone();
                                request["__agent_tool"] = value["tool"].clone();
                                request
                            },
                            reply,
                        ))
                        .is_err()
                    {
                        break;
                    }
                    match rx.recv() {
                        Ok(mut result) => {
                            result["id"] = json!(local_id);
                            if writeln!(stdin, "{result}").is_err() {
                                break;
                            }
                        }
                        Err(_) => break,
                    }
                } else if tx.send(Event::Message(turn, value)).is_err() {
                    break;
                }
            }
            drop(stdin);
            if let Ok(mut c) = child.lock() {
                let _ = c.kill();
                let _ = c.wait();
            }
            let _ = tx.send(Event::Exit(turn));
        });
        self.running = true;
        self.stopping = false;
        self.stop_requested = false;
        panel.agent_active = true;
        self.turn_tool_failed = false;
        self.status = "AI processing | starting connector".into();
        self.log(format!("You: {}", self.input.text));
        self.input.load(String::new());
        true
    }
    fn stop(&mut self, panel: &mut Panel) {
        if !self.running && self.child.is_none() {
            self.status = if self.stopping {
                "Stopping | waiting for the AI process to close".into()
            } else {
                "Nothing to stop".into()
            };
            return;
        }
        let pending = panel.cancel_agent();
        let cancelled = self.tool_requests.len() + self.waiting.len() + pending;
        let child = self.child.take();
        let had_process = child.is_some();
        if let Some(child) = child {
            if let Ok(child) = child.lock() {
                terminate_process(child.id());
            }
            let watchdog = child.clone();
            std::thread::spawn(move || {
                for _ in 0..50 {
                    if let Ok(mut child) = watchdog.lock() {
                        if child.try_wait().ok().flatten().is_some() {
                            return;
                        }
                    }
                    std::thread::sleep(std::time::Duration::from_millis(100));
                }
                if let Ok(mut child) = watchdog.lock() {
                    let _ = child.kill();
                    let _ = child.wait();
                }
            });
        }
        for (_, reply) in self.waiting.drain() {
            let _ = reply.send(json!({"ok":false,"error":"AI turn stopped"}));
        }
        for (_, reply) in self.tool_requests.drain(..) {
            let _ = reply.send(json!({"ok":false,"error":"AI turn stopped"}));
        }
        self.pending_ui = None;
        self.ui_poll = std::time::Instant::now() - std::time::Duration::from_secs(1);
        self.running = false;
        self.stop_requested = had_process;
        self.stopping = had_process;
        // A partially completed tool chain is not valid model history.
        self.history = json!([]);
        self.status = format!(
            "Stopped by user | {cancelled} pending reply/queue entries cleared; dispatched tool outcomes may be unknown | {} | tools/terminals may continue; inspect Techne | completed file changes remain saved",
            if had_process { "closing AI process" } else { "no process was running" }
        );
    }

    pub fn update(&mut self, panel: &mut Panel, page: &str) {
        if let Some((settings, expected)) = self.entry_save.take() {
            self.queue_settings(panel, serde_json::to_string_pretty(&settings).unwrap()+"\n", expected, false);
        }
        for (id, result) in panel.take_agent_replies() {
            if id == UI_WRITE_ID {
                if let Some(write) = self.pending_ui.take() {
                    if write.root == panel.root {
                        if result["ok"] == true {
                            self.ui = parse_settings(&write.text).ok();
                            self.field_expected = Some(write.text.clone());
                            let matches_draft =
                                parse_settings(&self.field_input.text).ok() == self.ui;
                            self.ui_text = Some(write.text);
                            self.ui_error.clear();
                            self.status = "Socius text settings saved".into();
                            if write.close_field
                                && matches_draft
                                && self.field == Some(Field::Macros)
                            {
                                self.field = None;
                            }
                        } else {
                            self.ui_error = result["error"]
                                .as_str()
                                .unwrap_or("Settings write failed")
                                .into();
                            self.macro_error = self.ui_error.clone();
                            self.status = format!("Settings not saved: {}", self.ui_error);
                            self.ui_poll =
                                std::time::Instant::now() - std::time::Duration::from_secs(1);
                        }
                    }
                }
                continue;
            }
            if let Some(tx) = self.waiting.remove(&id) {
                let _ = tx.send(result);
            }
        }
        self.refresh_settings(panel, false);
        while let Ok(event) = self.rx.try_recv() {
            match event {
                Event::Copied(root, cut, result) => {
                    if root != panel.root {
                        continue;
                    }
                    match result {
                        Ok(()) => {
                            if let Some((field, expected)) = cut {
                                if self.field == field {
                                    let e = if field.is_some() {
                                        &mut self.field_input
                                    } else {
                                        &mut self.input
                                    };
                                    if e.selected_all && e.text == expected {
                                        e.insert("");
                                    }
                                }
                            }
                            self.status = "Copied to system clipboard".into();
                        }
                        Err(error) => self.status = error,
                    }
                }
                Event::Paste(field, result) if field == self.field && self.text_input_active() => match result {
                    Ok(s) => {
                        if self.field.is_some() {
                            self.field_input.insert(s.trim());
                        } else {
                            self.input.insert(&s);
                        }
                    }
                    Err(e) => self.status = e,
                },
                Event::Request(id, request, reply) => {
                    if (id >> 32) != self.turn || !self.running {
                        let _ = reply.send(json!({"ok":false,"error":"AI turn stopped"}));
                        continue;
                    }
                    let tool = request["__agent_tool"].as_str().unwrap_or("");
                    let command = request["command"].as_str().unwrap_or("");
                    if !self.permits_rpc(tool, command) {
                        let _ = reply.send(json!({"ok":false,"error":"AI tool permission denied"}));
                        continue;
                    }
                    self.status = format!("Waiting for tool: {}{}", tool,
                        request["id"].as_str().map(|id| format!(" ({id})")).unwrap_or_default());
                    if matches!(command, "tool_run" | "tool_status" | "tool_stop" | "tool_inspect" | "terminal_start" | "terminal_write" | "terminal_read" | "terminal_status" | "terminal_list" | "terminal_interrupt" | "terminal_close" | "terminal_resize") {
                        let mut request = request.clone();
                        request["__world_root"] = json!(panel.root);
                        self.tool_requests.push((request, reply));
                    } else if request["command"] == "context" {
                        let _ = reply
                            .send(json!({"ok":true,"result":self.visible_context(panel,page)}));
                    } else {
                        self.waiting.insert(id, reply);
                        panel.queue_agent(id, request);
                    }
                }
                Event::Exit(turn) if turn == self.turn => {
                    if self.stop_requested {
                        self.status = "Stopped by user | AI process closed | dispatched outcomes may be unknown; tools/terminals may continue; inspect Techne | completed file changes remain saved | ready for next message".into();
                    } else if self.running {
                        self.status =
                            "AI connection interrupted | no end event received | ready to retry".into();
                        self.history = json!([]);
                    }
                    self.running = false;
                    self.stopping = false;
                    self.stop_requested = false;
                    panel.agent_active = false;
                    self.child = None;
                }
                Event::AnswerCopied(session, ticket, label, result) => self.finish_copy(&session, ticket, label, result),
                Event::Message(turn, v) if turn == self.turn && self.running => {
                    match v["event"].as_str().unwrap_or("") {
                        "recovery" => self.log(format!("Model recovery: attempt {}/{} failed; retry={}; diagnostic={}",v["attempt"],v["max_attempts"],v["will_retry"],v["diagnostic_path"])),
                        "status" => {
                            self.status = format!("AI processing | {}", v["text"].as_str().unwrap_or(""));
                            // The connector owns the turn budget; the UI only mirrors its report.
                            // Both live shapes ("time budget: 895s of 900s remain; ..." and
                            // "time budget: unlimited; elapsed 5s; ...") are recognized. A
                            // finite report starts the countdown; unlimited keeps the turn's
                            // own start time for elapsed display, never a fake zero countdown.
                            if let Some(budget) = v["text"].as_str().filter(|_|self.turn_budget.is_none()).and_then(parse_turn_budget) {
                                self.turn_budget = Some(budget);
                                if matches!(budget, TurnBudget::Limited { .. }) {
                                    self.turn_budget_since = std::time::Instant::now();
                                }
                            }
                        },
                        "assistant" => {
                            let text = v["text"].as_str().unwrap_or("");
                            self.answer = Some(text.to_owned());
                            self.copy_ticket = self.copy_ticket.wrapping_add(1);
                            self.copy_status.clear();
                            self.log(format!("{}: {}", self.config["name"].as_str().unwrap_or("AI"), text));
                        },
                        "tool_result" => {
                            let r = &v["result"];
                            let ok = r["ok"] == true
                                && !r["result"]["sync_error"].is_string()
                                && r["result"]["state"] != "failed"
                                && !r["result"]["exit_code"].as_i64().is_some_and(|code| code != 0);
                            self.turn_tool_failed |= !ok;
                            self.status = if ok {
                                format!("AI processing | tool {} completed", v["name"].as_str().unwrap_or(""))
                            } else {
                                format!("Tool failed: {} | AI processing", v["name"].as_str().unwrap_or(""))
                            };
                            let detail = if let Some(state) = r["result"]["state"].as_str() {
                                format!("{state}; exit_code={}", r["result"]["exit_code"])
                            } else if ok {
                                if r["result"]["sync_error"].is_string() {
                                    r["result"]["sync_error"].as_str().unwrap_or("").to_owned()
                                } else {
                                    r["result"]["source"]
                                        .as_str()
                                        .or_else(|| r["result"]["path"].as_str())
                                        .unwrap_or("completed")
                                        .to_owned()
                                }
                            } else {
                                r["error"].as_str().unwrap_or("failed").into()
                            };
                            self.log(format!(
                                "Tool {}: {} {}",
                                v["name"].as_str().unwrap_or(""),
                                if !ok { "ERROR" } else if r["result"]["state"].is_string() { "STATE" } else { "OK" },
                                detail
                            ));
                        }
                        "done" => {
                            self.history = v["history"].clone();
                            if let Some(text) = v["text"].as_str() { self.answer = Some(text.to_owned()); }
                            else if self.answer.is_none() { self.answer = Self::latest_answer(&self.history); }
                            self.status = if self.turn_tool_failed {
                                "AI turn ended | tool failure recorded; review log | ready for next message"
                            } else {
                                "AI turn ended | review result | ready for next message"
                            }.into();
                            self.running = false;
                            self.stopping = false;
                            self.stop_requested = false;
                            panel.agent_active = false;
                        }
                        "error" => {
                            self.log(format!("Error: {}", v["text"].as_str().unwrap_or("")));
                            let unknown = v["tool_outcome_unknown"] == true
                                || v["text"].as_str().is_some_and(|s|s.contains("outcome unknown") || s.contains("result delivery unknown"));
                            self.status = if unknown {
                                "AI paused | tool outcome unknown; inspect saved files and Techne; no automatic replay | ready for next message"
                            } else {
                                "AI turn failed; completed changes remain. Check the log. | ready for next message"
                            }.into();
                            self.history = json!([]);
                            self.running = false;
                            panel.agent_active = false;
                        }
                        _ => (),
                    }
                }
                _ => (),
            }
        }
    }
    pub fn key(
        &mut self,
        sym: Keysym,
        ctrl: bool,
        shift: bool,
        alt: bool,
        panel: &mut Panel,
        page: &str,
    ) {
        // Composition belongs to the native input method. Do not send, close
        // the panel, or edit the draft while it is choosing characters.
        if !self.ime_preedit.is_empty() {
            return;
        }
        if sym == Keysym::Escape {
            if self.menu.take().is_some() {
                return;
            }
            if self.field.take().is_none() {
                self.chat_open = false;
                self.chat_focus = false;
            }
            return;
        }
        if self.menu.is_some() {
            return;
        }
        let letter = sym.key_char().map(|c| c.to_ascii_lowercase());
        if self.field.is_none()
            && self.chat_open
            && ctrl
            && letter == Some('c')
            && (shift || self.log_focus)
        {
            if shift || self.log_selected_all || self.answer.is_none() {
                self.copy_conversation(panel);
            } else { self.copy_answer(); }
            return;
        }
        if self.field.is_none() && self.log_focus {
            if ctrl && letter == Some('a') {
                self.log_selected_all = true;
            }
            return;
        }
        if ctrl && matches!(letter, Some('c' | 'x')) {
            let e = if self.field.is_some() {
                &self.field_input
            } else {
                &self.input
            };
            if e.selected_all && !e.text.is_empty() {
                let text = e.text.clone();
                let cut = (letter == Some('x')).then(|| (self.field, text.clone()));
                self.copy_text(text, cut, panel);
            }
            return;
        }
        if self.field.is_none() && self.chat_open && ctrl {
            let delta = match sym.key_char() {
                Some('+' | '=') => Some(1),
                Some('-') => Some(-1),
                Some('0') => Some(0),
                _ => None,
            };
            if let Some(delta) = delta {
                let mut v = self.ui_settings();
                let scale = if delta == 0 {
                    3
                } else {
                    (self.font_scale() + delta).clamp(2, 4)
                };
                v["chat"]["font_scale"] = json!(scale);
                self.queue_settings(
                    panel,
                    serde_json::to_string_pretty(&v).unwrap() + "\n",
                    self.ui_text.clone(),
                    false,
                );
                return;
            }
        }
        if self.field == Some(Field::Macros)
            && self.pending_ui.as_ref().is_some_and(|p| p.close_field)
        {
            return;
        }
        if ctrl && sym.key_char().is_some_and(|c| c.eq_ignore_ascii_case(&'v')) {
            let tx = self.tx.clone();
            let field = self.field;
            std::thread::spawn(move || {
                let _ = tx.send(Event::Paste(field, super::workspace_panel::clipboard()));
            });
            return;
        }
        if self.field.is_some()
            && matches!(sym, Keysym::Return | Keysym::KP_Enter)
            && (!matches!(self.field, Some(Field::Personality | Field::Macros)) || ctrl)
        {
            let value = self.field_input.text.trim().to_owned();
            if self.field == Some(Field::Macros) {
                match parse_settings(&value) {
                    Ok(v) => {
                        let text = serde_json::to_string_pretty(&v).unwrap() + "\n";
                        self.queue_settings(panel, text, self.field_expected.clone(), true);
                    }
                    Err(e) => self.macro_error = e,
                }
                return;
            }
            match self.field.take().unwrap() {
                Field::Url => self.config["base_url"] = json!(value),
                Field::Model => self.config["model"] = json!(value),
                Field::Key => self.key = value,
                Field::Name => {
                    self.config["name"] = json!(if value.is_empty() { "AI".into() } else { value })
                }
                Field::Personality => self.config["personality"] = json!(value),
                Field::Macros => (),
            }
            self.save_config();
            return;
        }
        if ctrl && matches!(sym, Keysym::Return | Keysym::KP_Enter) {
            self.start(panel, page);
            return;
        }
        let e = if self.field.is_some() {
            &mut self.field_input
        } else {
            &mut self.input
        };
        if ctrl {
            match sym.key_char().map(|c| c.to_ascii_lowercase()) {
                Some('a') => e.selected_all = true,
                Some('z') if shift => e.redo(),
                Some('z') => e.undo(),
                Some('y') => e.redo(),
                _ => (),
            };
            return;
        }
        if alt {
            return;
        }
        match sym {
            Keysym::BackSpace => e.backspace(),
            Keysym::Delete => e.delete(),
            Keysym::Left => e.horizontal(false),
            Keysym::Right => e.horizontal(true),
            Keysym::Up => e.vertical(false),
            Keysym::Down => e.vertical(true),
            Keysym::Home => e.home_end(false),
            Keysym::End => e.home_end(true),
            Keysym::Return | Keysym::KP_Enter => e.newline(),
            Keysym::Tab => e.insert("    "),
            _ => {
                if let Some(c) = sym.key_char().filter(|c| !c.is_control()) {
                    e.insert(&c.to_string());
                }
            }
        }
        e.ensure_visible(
            if matches!(self.field, Some(Field::Personality | Field::Macros)) {
                7
            } else {
                3
            },
        );
    }

    pub fn dispatch_tools(&mut self, tools: &mut super::tool_panel::ToolPanel, panel: &Panel) {
        for (request, reply) in self.tool_requests.drain(..) {
            let result = if request["__world_root"] == panel.root {
                tools.rpc(&request, panel)
            } else {
                json!({"ok":false,"error":"World changed"})
            };
            let _ = reply.send(result);
        }
    }
    fn permission(&self, group: &str) -> bool {
        if group == "run" {
            return self.config["permissions"]["run"].as_bool() == Some(true);
        }
        match self.config.get("permissions") {
            None => true,
            Some(v) if v.is_object() => match v.get(group) {
                None => true,
                Some(b) => b.as_bool().unwrap_or(false),
            },
            _ => false,
        }
    }
    fn permits_rpc(&self, tool: &str, command: &str) -> bool {
        let (group, commands): (&str, &[&str]) = match tool {
            "terminal_start" => ("run", &["terminal_start"]),
            "terminal_write" => ("run", &["terminal_write"]),
            "terminal_interrupt" => ("run", &["terminal_interrupt"]),
            "terminal_close" => ("run", &["terminal_close"]),
            "terminal_resize" => ("run", &["terminal_resize"]),
            "terminal_read" => ("read", &["terminal_read"]),
            "terminal_status" => ("read", &["terminal_status"]),
            "terminal_list" => ("read", &["terminal_list"]),
            "context" => ("read", &["context"]),
            "list_tools" => ("read", &["read_text"]),
            "inspect_tool" => ("read", &["tool_inspect"]),
            "register_tool" => ("write", &["tool_inspect", "read_text", "write_text"]),
            "tool_status" => ("read", &["tool_status"]),
            "run_tool" => ("run", &["tool_run", "tool_status"]),
            "stop_tool" => ("run", &["tool_stop"]),
            "list_files" => ("read", &["list"]),
            "list_entry_nodes" => ("read", &["node_entries"]),
            "find_nodes" => ("read", &["node_search"]),
            "get_node" => ("read", &["node_get"]),
            "walk_nodes" => ("read", &["node_walk"]),
            "rebuild_node_index" => ("read", &["node_index"]),
            "list_directory" => ("read", &["directory"]),
            "read_document" => ("read", &["read"]),
            "read_text" => ("read", &["read_text"]),
            "search_text" => ("read", &["directory", "read_text"]),
            "search_notes" => ("read", &["directory", "read_text", "node_index"]),
            "create_document" => ("write", &["create"]),
            "adopt_file" => ("write", &["adopt"]),
            "write_document" | "patch_document" => ("write", &["read", "save"]),
            "sync_document" => ("write", &["sync"]),
            "write_text" | "patch_text" => ("write", &["read_text", "write_text"]),
            "delete_document" => ("delete", &["read", "delete"]),
            "delete_text" => ("delete", &["read_text", "delete_text"]),
            "import_document" => ("import", &["import"]),
            _ => return false,
        };
        self.permission(group) && commands.contains(&command)
    }
    fn runtime_identity(root: &str) -> Value {
        static ID: std::sync::OnceLock<Value> = std::sync::OnceLock::new();
        let mut value=ID.get_or_init(|| {
            let pid=std::process::id();
            let executable=std::fs::read_link(format!("/proc/{pid}/exe")).ok();
            let hash=Command::new("sha256sum").arg(format!("/proc/{pid}/exe")).output().ok()
                .filter(|r|r.status.success()).and_then(|r|String::from_utf8(r.stdout).ok())
                .and_then(|s|s.split_whitespace().next().map(str::to_owned))
                .filter(|s|s.len()==64 && s.bytes().all(|b|b.is_ascii_hexdigit()));
            json!({"host_pid":pid,"executable":executable,"executable_sha256":hash,
                "project":std::env::current_dir().ok(),"workspace_backend":"embedded Rust Workspace in this host",
                "capability":"transport-recovery-20261024-v2","cross_worker_resume":false})
        }).clone();
        value["world"]=json!(root);value
    }
    fn visible_context(&self, panel: &Panel, page: &str) -> Value {
        if self.permission("read") {
            let mut context = panel.context(page);
            context["runtime"] = Self::runtime_identity(&panel.root);
            context["techne"] = json!({"registry_path":yohaku::techne::REGISTRY_PATH,"instructions":"list_tools, run_tool, tool_status, stop_tool; only the registered offline world sandbox is supported. Startup request is not success; inspect status."});
            context["development"] = json!({"guide":"memory/yohaku-development.md", "record":"memory/development.json",
                "instructions":"If prepared, read the guide and edit the Loom-managed source copy under files/yohaku-default. Registered yohaku-check/test/build/preview tools are available after human preparation. They do not replace or publish the host app."});
            context["socius"] = json!({"settings_path":SETTINGS_PATH,"settings":self.ui_settings(),"error":self.ui_error,
                "viewport":{"width":self.viewport.0,"height":self.viewport.1,"sidebar":self.viewport.2},
                "chat_bounds":self.chat_geometry(self.viewport.0,self.viewport.1,self.viewport.2),
                "instructions":"Read settings_path using read_text, then write_text with its revision to configure macros or chat. These settings hot reload. No automatic sending or code execution."});
            context
        } else {
            json!({"root":panel.root,"page":page,"read_permission":false})
        }
    }
    fn ui_settings(&self) -> Value {
        self.ui.clone().unwrap_or_else(|| defaults(self.macros()))
    }
    fn font_scale(&self) -> i32 {
        self.ui
            .as_ref()
            .and_then(|v| v["chat"]["font_scale"].as_i64())
            .unwrap_or(3) as i32
    }
    fn queue_settings(
        &mut self,
        panel: &mut Panel,
        text: String,
        expected: Option<String>,
        close_field: bool,
    ) -> bool {
        if self.pending_ui.is_some() {
            self.macro_error = "Wait for the current settings save".into();
            return false;
        }
        if let Err(e) = parse_settings(&text) {
            self.macro_error = e;
            return false;
        }
        self.pending_ui = Some(SettingsWrite {
            root: panel.root.clone(),
            text: text.clone(),
            close_field,
        });
        panel.queue_agent(UI_WRITE_ID,json!({"command":"write_text","path":SETTINGS_PATH,"expected":expected,"text":text,"__world_root":panel.root}));
        self.status = "Saving Socius text settings...".into();
        true
    }
    fn refresh_settings(&mut self, panel: &mut Panel, force: bool) {
        let changed = self.ui_root != panel.root;
        if !changed
            && (!force && self.ui_poll.elapsed() < std::time::Duration::from_millis(400)
                || self.gesture.is_some()
                || self.pending_ui.is_some())
        {
            return;
        }
        self.ui_poll = std::time::Instant::now();
        if changed {
            self.ui = None;
            self.ui = Some(defaults(self.macros()));
            self.ui_initialized = false;
            self.ui_root = panel.root.clone();
            self.ui_text = None;
            self.ui_error.clear();
            self.gesture = None;
            self.menu = None;
            if self.field == Some(Field::Macros) {
                self.field = None;
            }
        }
        match read_settings(std::path::Path::new(&panel.root)) {
            Ok(Some(text)) => {
                self.ui_initialized = true;
                match parse_settings(&text) {
                    Ok(v) => {
                        self.ui = Some(v);
                        self.ui_text = Some(text);
                        self.ui_error.clear();
                    }
                    Err(e) => {
                        self.ui_text = Some(text);
                        self.ui_error = e;
                    }
                }
            }
            Ok(None) => {
                self.ui_text = None;
                if !self.ui_initialized
                    && std::path::Path::new(&panel.root).is_dir()
                    && self.pending_ui.is_none()
                {
                    self.ui_initialized = true;
                    self.queue_settings(
                        panel,
                        serde_json::to_string_pretty(&self.ui_settings()).unwrap() + "\n",
                        None,
                        false,
                    );
                }
            }
            Err(e) => self.ui_error = e,
        }
    }
    fn macros(&self) -> Value {
        if let Some(ui) = &self.ui {
            return ui["macros"].clone();
        }
        match self.config.get("ai_macros") {
            None => default_macros(),
            Some(v) if validate_macros(v).is_ok() => v.clone(),
            Some(_) => json!([]), // Invalid bindings never grant unintended actions.
        }
    }
    fn run_macro(&mut self, item: &Value, context: &Value) {
        self.menu = None;
        let was_open = self.chat_open && self.chat_focus;
        let previous_session = self.session_id.clone();
        if item["action"] != "close_chat" {
            if let Err(error) = self.open_default(context["root"].as_str().unwrap_or("")) {
                self.route_error = Some(error.clone());
                self.status = error;
                self.chat_open = true;
                return;
            }
        }
        match item["action"].as_str().unwrap_or("open_chat") {
            "close_chat" => {
                self.chat_open = false;
                self.chat_focus = false;
                return;
            }
            "toggle_chat" if was_open && previous_session == self.session_id => { self.chat_open = false; self.chat_focus = false; },
            "toggle_chat" => (),
            _ => {
                self.chat_open = true;
                self.chat_focus = true;
            }
        }
        if self.chat_open {
            let prompt = expand_prompt(item["prompt"].as_str().unwrap_or(""), context);
            if !prompt.is_empty() {
                self.input.selected_all = false;
                self.input.cursor = self.input.text.len();
                if !self.input.text.is_empty() {
                    self.input.insert("\n");
                }
                self.input.insert(&prompt);
                self.input.ensure_visible(3);
            }
        }
    }
    pub fn shortcut(
        &mut self,
        sym: Keysym,
        ctrl: bool,
        shift: bool,
        alt: bool,
        pressed: bool,
        panel: &Panel,
        page: &str,
    ) -> bool {
        if self.field.is_some() {
            return false;
        }
        let Some(c) = sym.key_char() else {
            return false;
        };
        let macros = self.macros();
        let found = macros.as_array().unwrap().iter().find(|m| {
            m["shortcut"].as_str().and_then(parse_shortcut)
                == Some((ctrl, shift, alt, c.to_ascii_lowercase()))
        });
        if let Some(item) = found {
            if pressed {
                self.run_macro(item, &self.visible_context(panel, page));
            }
            true
        } else {
            false
        }
    }
    pub fn right_click(
        &mut self,
        x: f64,
        y: f64,
        width: i32,
        height: i32,
        panel: &Panel,
        page: &str,
    ) -> bool {
        if self.field.is_some() {
            return true;
        }
        let macros = self.macros();
        let items = macros.as_array().unwrap();
        let context = self.visible_context(panel, page);
        if let Some(item) = items.iter().find(|m| m["trigger"] == "right_click") {
            self.run_macro(item, &context);
            return true;
        }
        let items: Vec<Value> = items
            .iter()
            .filter(|m| m["trigger"] == "context_menu")
            .cloned()
            .collect();
        if items.is_empty() {
            self.menu = None;
            return false;
        }
        let rows = items.len().min(((height - 8).max(32) / 32) as usize);
        self.menu = Some(MacroMenu {
            x: x as i32,
            y: y as i32,
            items,
            context,
            top: 0,
            rows,
        });
        // Geometry also clamps when the window is resized after invocation.
        let _ = width;
        true
    }
    pub fn takes_keyboard(&self) -> bool {
        self.menu.is_some() || self.field.is_some() || (self.chat_open && self.chat_focus)
    }

    fn copy_text(&mut self, text: String, cut: Option<(Option<Field>, String)>, panel: &Panel) {
        if text.is_empty() {
            return;
        }
        let tx = self.tx.clone();
        let root = panel.root.clone();
        std::thread::spawn(move || {
            let result = super::host_clipboard::write_verified(&text);
            let _ = tx.send(Event::Copied(root, cut, result));
        });
    }
    pub fn clear_preedit(&mut self) {
        self.ime_preedit.clear();
    }
    pub fn ime(&mut self, event: ::winit::event::Ime) {
        use ::winit::event::Ime;
        if !self.text_input_active() {
            self.clear_preedit();
            return;
        }
        match event {
            Ime::Preedit(text, _) => self.ime_preedit = text,
            Ime::Commit(text) => {
                self.clear_preedit();
                let text: String = text
                    .chars()
                    .filter(|c| !c.is_control() || matches!(c, '\n' | '\t'))
                    .collect();
                if !text.is_empty() {
                    let e = if self.field.is_some() {
                        &mut self.field_input
                    } else {
                        &mut self.input
                    };
                    e.insert(&text);
                    e.ensure_visible(
                        if matches!(self.field, Some(Field::Personality | Field::Macros)) {
                            7
                        } else {
                            3
                        },
                    );
                }
            }
            Ime::Disabled => self.clear_preedit(),
            Ime::Enabled => (),
        }
    }
    fn text_input_active(&self) -> bool {
        self.menu.is_none()
            && (self.field.is_some() || (self.chat_open && self.chat_focus && !self.log_focus))
            && !(self.field == Some(Field::Macros)
                && self.pending_ui.as_ref().is_some_and(|p| p.close_field))
    }
    pub fn ime_cursor_area(
        &self,

        height: i32,
        sidebar: i32,
    ) -> Option<(i32, i32, i32)> {
        if !self.text_input_active() {
            return None;
        }
        let (e, x, y, w, scale, rows) = if let Some(field) = self.field {
            let w = (width - sidebar - 40).clamp(240, 620);
            let multiline = matches!(field, Field::Personality | Field::Macros);
            let h = if multiline { 320 } else { 170 };
            (
                &self.field_input,
                sidebar + (width - sidebar - w) / 2 + 20,
                (height - h) / 2 + 62,
                w - 40,
                2,
                if multiline { 7 } else { 1 },
            )
        } else {
            let (left, top, w, h) = self.chat_geometry(width, height, sidebar);
            let layout = ChatLayout::new(top, h, self.font_scale());
            (
                &self.input,
                left + 24,
                layout.input_y + 6,
                w - 48,
                layout.scale,
                3,
            )
        };
        let (row, col) = e.row_column();
        let starts = e.starts();
        let before: String = e.line(&starts, row).chars().take(col).collect();
        let advance = super::shell_text_width(&before, scale).min((w - 8).max(0) as f32) as i32;
        let line = scale * 8 + 8;
        Some((
            (x + advance).clamp(0, width.saturating_sub(1)),
            (y + row.saturating_sub(e.top).min(rows - 1) as i32 * line)
                .clamp(0, height.saturating_sub(1)),

    fn chat_geometry(
        &self,
        screen_width: i32,
        screen_height: i32,
        sidebar: i32,
    ) -> (i32, i32, i32, i32) {
        let v = self.ui_settings();
        let chat = &v["chat"];
        let available = (screen_width - sidebar - 16).max(0);
        let w = (chat["width"].as_i64().unwrap_or(880) as i32).min(available);
        let h = (chat["height"].as_i64().unwrap_or(720) as i32).min((screen_height - 16).max(0));
        let x = chat["x"]
            .as_i64()
            .map(|n| n as i32)
            .unwrap_or(screen_width - w - 8)
            .clamp(sidebar + 8, (screen_width - w - 8).max(sidebar + 8));
        let y = chat["y"]
            .as_i64()
            .map(|n| n as i32)
            .unwrap_or(screen_height - h - 8)
            .clamp(8, (screen_height - h - 8).max(8));
        (x, y, w, h)
    }
    pub fn pointer_motion(
        &mut self,
        x: f64,
        y: f64,
        width: i32,
        height: i32,
        sidebar: i32,
    ) -> bool {
        let Some(g) = &self.gesture else {
            return false;
        };
        let (left, top, w, h) = g.bounds;
        let dx = x as i32 - g.x;
        let dy = y as i32 - g.y;
        let (max_w, max_h) = ((width - sidebar - 16).max(0), (height - 16).max(0));
        let (min_w, min_h) = (480.min(max_w), 420.min(max_h));
        let mut l = left;
        let mut t = top;
        let mut r = left + w;
        let mut b = top + h;
        if g.edges == 0 {
            l = (left + dx).clamp(sidebar + 8, (width - w - 8).max(sidebar + 8));
            t = (top + dy).clamp(8, (height - h - 8).max(8));
            r = l + w;
            b = t + h;
        } else {
            if g.edges & 1 != 0 {
                l = (left + dx).clamp(sidebar + 8, (r - min_w).max(sidebar + 8));
            }
            if g.edges & 2 != 0 {
                r = (left + w + dx).clamp(l + min_w, (width - 8).max(l + min_w));
            }
            if g.edges & 4 != 0 {
                t = (top + dy).clamp(8, (b - min_h).max(8));
            }
            if g.edges & 8 != 0 {
                b = (top + h + dy).clamp(t + min_h, (height - 8).max(t + min_h));
            }
        }
        let mut v = self.ui_settings();
        v["chat"]["x"] = json!(l.max(0));
        v["chat"]["y"] = json!(t.max(0));
        // Store legal preference sizes even when a temporary small window clamps them.
        v["chat"]["width"] = json!((r - l).max(480));
        v["chat"]["height"] = json!((b - t).max(420));
        self.ui = Some(v);
        true
    }
    pub fn pointer_release(&mut self, panel: &mut Panel) -> bool {
        let Some(g) = self.gesture.take() else {
            return false;
        };
        if g.root == panel.root {
            self.queue_settings(
                panel,
                serde_json::to_string_pretty(&self.ui_settings()).unwrap() + "\n",
                g.expected,
                false,
            );
        }
        true
    }
    pub fn hit_global(
        &self,
        x: f64,
        y: f64,
        screen_width: i32,
        screen_height: i32,
        sidebar: i32,
    ) -> bool {
        if self.field.is_some() {
            return true;
        }
        let (x, y) = (x as i32, y as i32);
        if self
            .menu
            .as_ref()
            .is_some_and(|m| inside(x, y, m.geometry(screen_width, screen_height)))
        {
            return true;
        }
        self.chat_open
            && inside(
                x,
                y,
                self.chat_geometry(screen_width, screen_height, sidebar),
            )
    }
    pub fn click_global(
        &mut self,
        x: f64,
        y: f64,
        screen_width: i32,
        screen_height: i32,
        sidebar: i32,
        panel: &mut Panel,
        page: &str,
    ) {
        self.clear_preedit();
        if self.field.is_some() {
            return;
        }
        let (x, y) = (x as i32, y as i32);
        if let Some(menu) = self.menu.take() {
            let bounds = menu.geometry(screen_width, screen_height);
            if inside(x, y, bounds) {
                let index = menu.top + ((y - bounds.1) / 32) as usize;
                if let Some(item) = menu.items.get(index) {
                    self.run_macro(item, &menu.context);
                }
                return;
            }
        }
        let (left, top, w, h) = self.chat_geometry(screen_width, screen_height, sidebar);
        if !self.chat_open || !inside(x, y, (left, top, w, h)) {
            self.chat_focus = false;
            return;
        }
        self.chat_focus = true;
        self.log_focus = false;
        self.log_selected_all = false;
        if y < top + 40 && x > left + w - 76 {
            self.chat_open = false;
            self.chat_focus = false;
            return;
        }
        let answer_x = left + w - if w >= 464 { 328 } else { 208 };
        if inside(x, y, (answer_x, top + 4, 124, 36)) {
            self.log_focus = true;
            self.copy_answer();
            return;
        }
        if w >= 464 && inside(x, y, (left + w - 192, top + 4, 108, 36)) {
            self.log_focus = true;
            self.log_selected_all = true;
            self.copy_conversation(panel);
            return;
        }
        let mut edges = 0;
        if x < left + 8 {
            edges |= 1;
        }
        if x >= left + w - 8 {
            edges |= 2;
        }
        if y < top + 8 {
            edges |= 4;
        }
        if y >= top + h - 8 {
            edges |= 8;
        }
        if (edges != 0 || y < top + 44) && self.pending_ui.is_none() {
            self.gesture = Some(Gesture {
                x,
                y,
                bounds: (left, top, w, h),
                edges,
                expected: self.ui_text.clone(),
                root: panel.root.clone(),
            });
            return;
        }
        let bottom = top + h;
        if (bottom - 50..bottom - 14).contains(&y) {
            match x - left {
                12..=185 => self.primary_action(panel, page),
                192..=295 if !self.running && !self.stopping => {
                    self.history = json!([]);
                    self.logs.clear();
                    self.answer = None;
                    self.copy_ticket = self.copy_ticket.wrapping_add(1);
                    self.copy_status.clear();
                    self.status = "New conversation".into();
                }
                _ => (),
            }
        } else {
            let layout = ChatLayout::new(top, h, self.font_scale());
            if (layout.log_y..layout.log_y + layout.log_h).contains(&y) {
                self.log_focus = true;
                self.status =
                    "Answer: Ctrl+C copies raw answer; Ctrl+A then Ctrl+C or Ctrl+Shift+C copies chat"
                        .into();
                return;
            }
            if !(layout.input_y..layout.input_y + layout.input_h).contains(&y) {
                return;
            }
            let row = self.input.top + ((y - layout.input_y - 6).max(0) / layout.line) as usize;
            let starts = self.input.starts();
            let row = row.min(starts.len() - 1);
            let col = super::shell_text_column(
                self.input.line(&starts, row),
                layout.scale,
                (x - left - 24).max(0) as f32,
            );
            self.input.place(row, col);
        }
    }

                    item["name"].as_str().unwrap_or(""),
                    x + 12,
                    y + n as i32 * 32 + 6,
                    2,
                    x + w - 12,
                    (dark, white),
                )?;
            }
        }
        if self.field.is_some() {
            self.draw_field(frame, screen_width, screen_height, sidebar)?;
        }
        Ok(())
    }
    fn draw_editor<F: Frame>(
        &self,
        frame: &mut F,
        e: &Editor,
        x: i32,
        y: i32,
        width: i32,
        rows: usize,
        masked: bool,
        scale: i32,
    ) -> Result<(), F::Error> {
        let (_, bg, dark, _, blue) = palette();
        let starts = e.starts();
        let (cursor_row, cursor_col) = e.row_column();
        for row in e.top..(e.top + rows).min(starts.len()) {
            let text = e.line(&starts, row);
            let text = if masked {
                "*".repeat(text.chars().count())
            } else {
                text.into()
            };
            let skip = if row == cursor_row {
                let mut skip = 0;
                while super::shell_text_width(
                    &text
                        .chars()
                        .skip(skip)
                        .take(cursor_col.saturating_sub(skip))
                        .collect::<String>(),
                    scale,
                ) > (width - 8) as f32
                    && skip < cursor_col
                {
                    skip += 1;
                }
                skip
            } else {
                0
            };
            let visible: String = text.chars().skip(skip).collect();
            let y = y + (row - e.top) as i32 * (scale * 8 + 8);
            let color = if e.selected_all { blue } else { bg };
            if e.selected_all {
                frame.clear(
                    blue,
                    &[rect(
                        (x, y),
                        (
                            super::shell_text_width(&visible, scale)
                                .min(width as f32)
                                .max(8.) as i32,
                            scale * 8 + 8,
                        ),
                    )],
                )?;
            }
            super::draw_shell_text_on(frame, &visible, x, y, scale, x + width, (dark, color))?;
            if row == cursor_row && !e.selected_all {
                let before: String = text
                    .chars()
                    .skip(skip)
                    .take(cursor_col.saturating_sub(skip))
                    .collect();
                let caret = x + super::shell_text_width(&before, scale) as i32;
                frame.clear(dark, &[rect((caret, y + 2), (1, scale * 8 + 4))])?;
            }
        }
        Ok(())
    }
    fn draw_field<F: Frame>(
        &self,
        frame: &mut F,
        screen_width: i32,
        screen_height: i32,
        sidebar: i32,
    ) -> Result<(), F::Error> {
        let Some(field) = self.field else {
            return Ok(());
        };
        let (white, bg, dark, muted, _) = palette();
        let multiline = matches!(field, Field::Personality | Field::Macros);
        let w = (screen_width - sidebar - 40).clamp(240, 620);
        let h = if multiline { 320 } else { 170 };
        let x = sidebar + (screen_width - sidebar - w) / 2;
        let y = (screen_height - h) / 2;
        frame.clear(muted, &[rect((x - 1, y - 1), (w + 2, h + 2))])?;
        frame.clear(white, &[rect((x, y), (w, h))])?;
        let title = match field {
            Field::Url => "API base URL",
            Field::Model => "Model name",
            Field::Key => "API key (memory only)",
            Field::Name => "AI name",
            Field::Personality => "AI personality and instructions",
            Field::Macros => "Socius world text: macros and chat (JSON)",
        };
        super::draw_shell_text_on(frame, title, x + 16, y + 14, 2, x + w - 16, (dark, white))?;
        frame.clear(
            bg,
            &[rect(
                (x + 12, y + 54),
                (w - 24, if multiline { 192 } else { 40 }),
            )],
        )?;
        self.draw_editor(
            frame,
            &self.field_input,
            x + 20,
            y + 62,
            w - 40,
            if multiline { 7 } else { 1 },
            field == Field::Key,
            2,
        )?;
        super::draw_shell_text_on(
            frame,
            if multiline {
                "Ctrl+Enter save / Enter newline / Escape cancel"
            } else {
                "Enter save / Escape cancel"
            },
            x + 16,
            y + h - 60,
            2,
            x + w - 16,
            (muted, white),
        )?;
        super::draw_shell_text_on(
            frame,
            if !self.ime_preedit.is_empty() {
                if field == Field::Key {
                    "Composing..."
                } else {
                    &self.ime_preedit
                }
            } else if self.macro_error.is_empty() {
                "Ctrl+A all / Ctrl+C copy / Ctrl+V paste"
            } else {
                &self.macro_error
            },
            x + 16,
            y + h - 30,
            2,
            x + w - 16,
            (muted, white),
        )?;
        Ok(())
    }
}
impl Drop for AgentPanel {
    fn drop(&mut self) {
        if let Some(c) = self.child.take() {
            if let Ok(mut c) = c.lock() {
                let _ = c.kill();
            }
        }
        self.waiting.clear();
    }
}

fn button<F: Frame>(
    frame: &mut F,
    x: i32,
    y: i32,
    w: i32,
    label: &str,
    right: i32,
    fg: Color32F,
    bg: Color32F,
) -> Result<(), F::Error> {
    if x + w > right {
        return Ok(());
    }
    frame.clear(bg, &[rect((x, y), (w, 36))])?;
    super::draw_shell_text_on(frame, label, x + 10, y + 6, 2, x + w - 4, (fg, bg))
}

pub(super) fn wrap(text: &str, width: f32) -> Vec<String> {
    wrap_scaled(text, width, 2)
}
fn wrap_scaled(text: &str, width: f32, scale: i32) -> Vec<String> {
    let mut out = Vec::new();
    let mut line = String::new();
    let mut w = 0.;
    for c in text.chars() {
        let advance = super::shell_text_width(&c.to_string(), scale);
        if w + advance > width && !line.is_empty() {
            out.push(std::mem::take(&mut line));
            w = 0.;
        }
        line.push(c);
        w += advance;
    }
    out.push(line);
    out
}

    #[test]
    pub(crate) fn window_protocol_tests_turn_budget_line_counts_down_and_hides() {
        let now = std::time::Instant::now();
        let reported = "AI processing | time budget: 895s of 900s remain; elapsed 5s";
        let budget = TurnBudget::Limited { remaining: 895, total: 900 };
        // The first frame preserves the report (895 left / 5 used) instead of
        // subtracting the report's elapsed time a second time.
        assert!(budget_status(reported.into(), Some(budget), now, true)
            .starts_with("budget 895s left of 900s (5s used) |"));
        let five = now - std::time::Duration::from_secs(5);
        assert!(budget_status(reported.into(), Some(budget), five, true)
            .starts_with("budget 890s left of 900s (10s used) |"));
        // Never negative, and no budget line without a report or after the turn ended.
        let expired = now - std::time::Duration::from_secs(895);
        assert!(budget_status(reported.into(), Some(budget), expired, true)
            .starts_with("budget 0s left of 900s (900s used) |"));
        assert_eq!(budget_status(reported.into(), Some(budget), now, false), reported);
        assert_eq!(budget_status(reported.into(), None, now, true), reported);
        // Unlimited shows elapsed turn time and never a fake zero countdown.
        let unlimited = "AI processing | time budget: unlimited; elapsed 0s";
        let twelve = now - std::time::Duration::from_secs(12);
        let shown = budget_status(unlimited.into(), Some(TurnBudget::Unlimited), twelve, true);
        assert!(shown.starts_with("budget unlimited; 12s elapsed |"), "{}", shown);
        assert!(!shown.contains("0s left"), "{}", shown);
        assert!(!shown.contains("elapsed 0s"), "{}", shown);
        assert_eq!(budget_status(unlimited.into(), Some(TurnBudget::Unlimited), twelve, false), unlimited);
    }

    #[test]
    pub(crate) fn window_protocol_tests_turn_budget_is_not_extended_by_later_reports() {
        let mut a = AgentPanel::new();
        let mut panel = render_panel();
        a.turn = 1;
        a.running = true;
        a.tx.send(Event::Message(1, json!({"event":"status",
            "text":"time budget: 900s of 900s remain; elapsed 0s; Stop remains available"}))).unwrap();
        a.update(&mut panel, "files");
        assert_eq!(a.turn_budget, Some(TurnBudget::Limited { remaining: 900, total: 900 }));
        // Long activity reports a smaller remaining time; the mirror keeps the first
        // report of the turn instead of restarting the countdown.
        a.tx.send(Event::Message(1, json!({"event":"status",
            "text":"time budget: 700s of 900s remain; elapsed 200s; Stop remains available"}))).unwrap();
        a.update(&mut panel, "files");
        assert_eq!(a.turn_budget, Some(TurnBudget::Limited { remaining: 900, total: 900 }));
        // After the turn ends the line is hidden. begin_turn() clears the mirror, so
        // the next connector report acquires a fresh remaining/total pair.
        a.tx.send(Event::Message(1, json!({"event":"done","history":[]}))).unwrap();
        a.update(&mut panel, "files");
        assert!(!a.running);
        assert_eq!(budget_status(a.status.clone(), a.turn_budget, a.turn_budget_since, a.running), a.status);
        assert_eq!(a.begin_turn(), 2);
        assert_eq!(a.turn_budget, None);
        a.running = true;
        a.tx.send(Event::Message(2, json!({"event":"status",
            "text":"time budget: 600s of 600s remain; elapsed 0s; Stop remains available"}))).unwrap();
        a.update(&mut panel, "files");
        assert_eq!(a.turn_budget, Some(TurnBudget::Limited { remaining: 600, total: 600 }));
    }
