struct ChatSession {
    answer: Option<String>,
    turn_tool_failed: bool,
    tool_summary: String,
    config: Value,
    key: String,
    input: Editor,
    logs: Vec<String>,
    log_offset: usize,
    status: String,
    history: Value,
    root: String,
}
impl AgentPanel {
    fn identity(&self) -> &str {
        if self.route_error.is_some() { return "Agent entry unavailable"; }
        if self.session_id.contains("\ncodex:") { "Codex / codex-local" } else { "Yohaku Agent / direct" }
    }
    fn select_session(&mut self, root: &str, backend: &str, config: Value) -> Result<(), String> {
        if self.running { return Err("Finish or stop the current turn before switching Agent".into()); }
        let key = if backend == "agent" { "agent".into() } else { format!("codex:{}", config["codex_entry"].as_str().ok_or("Missing Codex entry")?) };
        let identifier = format!("{root}\n{key}");
        if identifier != self.session_id {
            let saved = ChatSession {
                answer: self.answer.take(),
                turn_tool_failed: std::mem::take(&mut self.turn_tool_failed),
                tool_summary: std::mem::take(&mut self.tool_summary),
                config: std::mem::replace(&mut self.config, config.clone()),
                key: std::mem::take(&mut self.key), input: std::mem::take(&mut self.input),
                logs: std::mem::take(&mut self.logs), log_offset: self.log_offset,
                status: std::mem::take(&mut self.status), history: std::mem::replace(&mut self.history, json!([])),
                root: std::mem::replace(&mut self.history_root, root.into()),
            };
            if !self.session_id.is_empty() { self.sessions.insert(self.session_id.clone(), saved); }
            self.session_id = identifier;
            self.log_offset = 0;
            if let Some(saved) = self.sessions.remove(&self.session_id) {
                self.answer = saved.answer;
                self.turn_tool_failed = saved.turn_tool_failed;
                self.tool_summary = saved.tool_summary;
                self.config = saved.config; self.key = saved.key; self.input = saved.input;
                self.logs = saved.logs; self.log_offset = saved.log_offset; self.status = saved.status;
                self.history = saved.history; self.history_root = saved.root;
            }
        }
        if self.config != config {
            self.config = config;
            self.key.clear();
            self.history = json!([]);
            self.answer = None;
            self.status = "Entry configuration reloaded; new model history".into();
        }
        self.copy_ticket = self.copy_ticket.wrapping_add(1);
        self.copy_status.clear();
        self.route_error = None;
        self.clear_preedit();
        self.field = None;
        self.log_focus = false;
        self.log_selected_all = false;
        self.chat_open = true;
        self.chat_focus = true;
        Ok(())
    }
    fn world_entries(root: &str) -> Result<Value, String> {
        let text = read_settings(std::path::Path::new(root))?.ok_or("Missing memory/socius.json; configure chat.backend and chat.provider")?;
        parse_settings(&text)
    }
    fn open_default(&mut self, root: &str) -> Result<(), String> {
        let settings = Self::world_entries(root)?;
        let backend = settings["chat"]["backend"].as_str().ok_or("Missing chat.backend in memory/socius.json")?;
        let config = yohaku::socius::entry_config(&settings, &self.legacy_config, backend)?;
        self.select_session(root, backend, config)
    }
    pub fn connect_local_codex(&mut self, connection: Value, panel: &Panel) {
        let result = (|| {
            if connection["protocol"] != "codex-local" { return Err("Invalid Codex connector result".into()); }
            let mut config = self.legacy_config.clone();
            config.as_object_mut().ok_or("Invalid host configuration")?.remove("api_key");
            config.as_object_mut().unwrap().remove("credential_source");
            for key in ["protocol", "codex_entry"] { config[key] = connection[key].clone(); }
            config["name"] = json!("Codex");
            let settings=Self::world_entries(&panel.root)?;
            yohaku::socius::apply_task_limits(&settings,&mut config);
            self.select_session(&panel.root, "codex-local", config)
        })();
        match result {
            Ok(()) => self.status = "Codex session ready; Chat default is unchanged".into(),
            Err(error) => self.status = error,
        }
    }
    fn save_entry_config(&mut self) -> bool {
        if self.session_id.is_empty() { return false; }
        if self.config.get("permissions") != self.legacy_config.get("permissions") {
            let result = (|| -> Result<(), String> {
                let current = std::fs::read_to_string(&self.config_path).map_err(|error|error.to_string())?;
                let mut saved: Value = serde_json::from_str(&current).map_err(|error|error.to_string())?;
                if saved.get("permissions") != self.legacy_config.get("permissions") { return Err("Host permissions changed; reopen settings before saving".into()); }
                saved["permissions"] = self.config["permissions"].clone();
                std::fs::write(self.config_path.with_extension("update.json"),serde_json::to_vec_pretty(&saved).unwrap()).map_err(|error|error.to_string())?;
                std::fs::rename(self.config_path.with_extension("update.json"),&self.config_path).map_err(|error|error.to_string())?;
                self.legacy_config["permissions"] = self.config["permissions"].clone();
                Ok(())
            })();
            if let Err(error)=result { self.status=error; return true; }
        }
        if self.config["protocol"] == "codex-local" {
            self.status = "Codex session settings updated; account and Chat default unchanged".into();
            return true;
        }
        let result = (|| {
            let mut settings = Self::world_entries(&self.history_root)?;
            let mut provider = serde_json::Map::new();
            for key in ["protocol","base_url","model","api_key_env","credential_source","max_tokens","timeout","request_options","name","personality"] {
                if let Some(value) = self.config.get(key) { provider.insert(key.into(), value.clone()); }
            }
            settings["chat"]["provider"] = Value::Object(provider);
            yohaku::socius::entry_config(&settings,&self.legacy_config,"agent")?;
            self.entry_save = Some((settings, self.ui_text.clone()));
            Ok::<(),String>(())
        })();
        self.status = match result { Ok(()) => "Saving world Agent settings...".into(), Err(error) => error };
        true
    }
}

    // Both live connector shapes start with "time budget: " and may report either
    // a finite remaining/total pair or a distinct unlimited state. Parsing must keep
    // both finite integers distinct and never invent a zero countdown for unlimited.
    #[test]
    fn entry_turn_budget_parses_real_connector_status() {
        assert_eq!(parse_turn_budget("time budget: 900s of 900s remain; elapsed 0s; Stop remains available"),
            Some(TurnBudget::Limited { remaining: 900, total: 900 }));
        assert_eq!(parse_turn_budget("time budget: 895s of 900s remain; elapsed 5s; Stop remains available"),
            Some(TurnBudget::Limited { remaining: 895, total: 900 }));
        assert_eq!(parse_turn_budget("time budget: 7s of 7s remain; elapsed 0s; Stop remains available"),
            Some(TurnBudget::Limited { remaining: 7, total: 7 }));
        assert_eq!(parse_turn_budget("time budget: 86400s of 86400s remain; elapsed 0s; Stop remains available"),
            Some(TurnBudget::Limited { remaining: 86400, total: 86400 }));
        assert_eq!(parse_turn_budget("time budget: unlimited; elapsed 5s; source=worker default: unlimited"),
            Some(TurnBudget::Unlimited));
        // Non-budget status lines must not produce an invented budget.
        assert_eq!(parse_turn_budget("Model request attempt 2/3 failed: SSLEOFError; retry pending; Stop available"),None);
        assert_eq!(parse_turn_budget("No total turn limit selected; socket timeout=120s; Stop remains available"),None);
        // Every malformed piece below returns None instead of panicking or guessing.
        for bad in ["", "time budget: ", "time budget: of 900s", "time budget: 900",
            "time budget: 900m of 900m remain", "time budget: 0s of 0s remain",
            "time budget: -5s of 900s remain", "time budget: 901s of 900s remain",
            "time budget: 999999999999s of 999999999999s remain",
            "time budget: unlimitedly; elapsed 5s",
            "budget: 900s of 900s remain", "time budget 900s"] {
            assert_eq!(parse_turn_budget(bad),None,"{}",bad);
        }
    }
