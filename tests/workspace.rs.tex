#[test]
fn adopt_keeps_project_paths_and_loom_controls_edits_without_overwriting_sources() {
    let mut f = Fixture::new();
    let output = "files/yohaku-default/src/main.rs";
    let path = f.path(output);
    fs::create_dir_all(path.parent().unwrap()).unwrap();
    fs::write(&path, "fn main() { println!(\"before\"); }\n").unwrap();
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        fs::set_permissions(&path, fs::Permissions::from_mode(0o755)).unwrap();
    }
    let result = f
        .workspace
        .execute(&json!({"command":"adopt","path":output}))
        .unwrap();
    assert_eq!(result["output"], output);
    let source = result["source"].as_str().unwrap();
    let doc = f.workspace.read(source).unwrap();
    assert!(doc["body"].as_str().unwrap().contains("before"));
    f.workspace
        .save(
            source,
            doc["text"].as_str().unwrap(),
            "fn main() { println!(\"after\"); }\n",
            true,
        )
        .unwrap();
    assert!(fs::read_to_string(&path).unwrap().contains("after"));
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        assert_eq!(
            fs::metadata(&path).unwrap().permissions().mode() & 0o777,
            0o755
        );
    }
    let previous = fs::read(f.path(source)).unwrap();
    let again = f
        .workspace
        .execute(&json!({"command":"adopt","path":output}))
        .unwrap();
    assert_eq!(again["already_managed"], true);
    assert_eq!(fs::read(f.path(source)).unwrap(), previous);
    for invalid in [
        "../outside.rs",
        "memory/note.md",
        "files/.yohaku/file",
        "files/test.rs.tex",
    ] {
        assert!(
            f.workspace
                .execute(&json!({"command":"adopt","path":invalid}))
                .is_err()
        );
    }
    let other = f.path("files/other.py");
    fs::write(&other, "print(1)\n").unwrap();
    fs::write(f.path("files/other.py.tex"), "a human draft").unwrap();
    assert!(
        f.workspace
            .execute(&json!({"command":"adopt","path":"files/other.py"}))
            .is_err()
    );
    assert_eq!(
        fs::read_to_string(f.path("files/other.py.tex")).unwrap(),
        "a human draft"
    );
}

fn import_uses_real_loom_is_byte_exact_and_does_not_touch_or_execute_original() {
    let mut f = Fixture::new();

    assert!(f.path(result["source"].as_str().unwrap()).exists());

        .as_str()
        .unwrap()
        .to_owned();

    let value = f.workspace.delete(&source, &text).unwrap();
    assert_eq!(value["ok"], false);
    assert!(!value["result"]["retained"].as_array().unwrap().is_empty());
    assert_eq!(fs::read(f.path("files/b.py")).unwrap(), b"manual edit");
}

#[test]
fn concurrent_callers_keep_all_registry_entries_and_do_not_overwrite_each_other() {
    let f = Fixture::new();
    let mut children = Vec::new();
    for n in 0..4 {
        let mut child = Command::new(env!("CARGO_BIN_EXE_yohaku_workspace"))
            .args([
                "--root",
                f.workspace.root.to_str().unwrap(),
                "--loom",
                f.loom.to_str().unwrap(),
            ])
            .stdin(Stdio::piped())
            .stdout(Stdio::piped())
            .spawn()
            .unwrap();
        let request = json!({"command":"create","name":"shared.py","body":format!("caller {n}")});
        child
            .stdin
            .take()
            .unwrap()
            .write_all(request.to_string().as_bytes())
            .unwrap();
        children.push(child);
    }
    let mut outputs = Vec::new();
    for child in children {
        let result = child.wait_with_output().unwrap();
        assert!(
            result.status.success(),
            "{}",
            String::from_utf8_lossy(&result.stdout)
        );
        let value: Value = serde_json::from_slice(&result.stdout).unwrap();
        outputs
            .push(fs::read_to_string(f.path(value["result"]["output"].as_str().unwrap())).unwrap());
    }
    outputs.sort();
    assert_eq!(
        outputs,
        vec!["caller 0", "caller 1", "caller 2", "caller 3"]
    );
    let reopened = Workspace::open(&f.workspace.root, &f.loom).unwrap();
    assert_eq!(
        reopened.list().unwrap()["files"].as_array().unwrap().len(),
        4
    );
}

    f.workspace.execute(&req).unwrap();

    f.workspace.execute(&json!({"command":"write_text","path":"memory/notes.md","expected":"remember 🧸","text":"updated"})).unwrap();
    let listing = f
        .workspace
        .execute(&json!({"command":"directory","path":"memory"}))
        .unwrap();
    assert_eq!(listing["entries"][0]["path"], "memory/notes.md");
    let created = f.workspace.create("demo.py", "old").unwrap();
    let source = created["source"].as_str().unwrap();
    let old = f.workspace.read(source).unwrap()["text"]
        .as_str()
        .unwrap()
        .to_owned();
    let new = old.replace("old", "new");
    f.workspace
        .execute(&json!({"command":"write_text","path":source,"expected":old,"text":new}))
        .unwrap();
    assert_eq!(fs::read_to_string(f.path("files/demo.py")).unwrap(), "new");
    for path in [
        "files/demo.py",
        ".yohaku/files.json",
        "files/.loom/state.json",
        "../escape",
    ] {
        assert!(
            f.workspace
                .execute(&json!({"command":"write_text","path":path,"expected":null,"text":"bad"}))
                .is_err(),
            "{path}"
        );
    }
    f.workspace
        .execute(&json!({"command":"delete_text","path":source,"expected":new}))
        .unwrap();
    assert!(!f.path("files/demo.py").exists());
    f.workspace
        .execute(&json!({"command":"delete_text","path":"memory/notes.md","expected":"updated"}))
        .unwrap();

}
