//! The world-owned, human/agent editable Socius interaction settings.
use serde_json::{Value, json};
pub const SETTINGS_PATH: &str = "memory/socius.json";
pub const MAX_SETTINGS_BYTES: usize = 256 * 1024;
pub fn default_macros() -> Value {
    json!([{"name":"Chat","trigger":"context_menu","shortcut":"Ctrl+J","action":"open_chat","prompt":""}])
}
pub fn parse_shortcut(s: &str) -> Option<(bool, bool, bool, char)> {
    let mut ctrl = false;
    let mut shift = false;
    let mut alt = false;
    let mut key = None;
    for p in s.split('+') {
        match p.trim().to_ascii_lowercase().as_str() {
            "ctrl" if !ctrl => ctrl = true,
            "shift" if !shift => shift = true,
            "alt" if !alt => alt = true,
            p if p.len() == 1 && p.as_bytes()[0].is_ascii_alphanumeric() && key.is_none() => {
                key = p.chars().next()
            }
            _ => return None,
        }
    }
    if !ctrl && !alt {
        return None;
    }
    key.map(|k| (ctrl, shift, alt, k))
}
pub fn validate_macros(v: &Value) -> Result<(), String> {
    let items = v.as_array().ok_or("Expected a JSON array")?;
    if items.len() > 12 {
        return Err("At most 12 AI macros".into());
    }
    let mut shortcuts = Vec::new();
    let mut direct = false;
    for item in items {
        let obj = item.as_object().ok_or("Each macro must be an object")?;
        if obj
            .keys()
            .any(|k| !["name", "trigger", "shortcut", "action", "prompt"].contains(&k.as_str()))
        {
            return Err("Unknown macro field".into());
        }
        let name = item["name"].as_str().ok_or("Each macro needs a name")?;
        if name.trim().is_empty() || name.chars().count() > 80 || name.chars().any(char::is_control)
        {
            return Err("Macro name must contain 1-80 visible characters".into());
        }
        let trigger = item["trigger"].as_str().ok_or("Missing trigger")?;
        if !["context_menu", "right_click", "shortcut"].contains(&trigger) {
            return Err("Invalid trigger".into());
        }
        if trigger == "right_click" {
            if direct {
                return Err("Only one direct right-click macro".into());
            }
            direct = true;
        }
        if let Some(action) = item.get("action") {
            if !["open_chat", "toggle_chat", "close_chat"]
                .iter()
                .any(|a| action.as_str() == Some(a))
            {
                return Err("Invalid action".into());
            }
        }
        if let Some(prompt) = item.get("prompt") {
            if prompt.as_str().is_none_or(|p| p.len() > 16384) {
                return Err("Prompt must be text, at most 16 KiB".into());
            }
        }
        if let Some(shortcut) = item.get("shortcut").filter(|s| !s.is_null()) {
            let parsed = shortcut
                .as_str()
                .and_then(parse_shortcut)
                .ok_or("Shortcut: Ctrl/Alt + letter or digit")?;
            if shortcuts.contains(&parsed) {
                return Err("Duplicate shortcut".into());
            }
            shortcuts.push(parsed);
        } else if trigger == "shortcut" {
            return Err("Shortcut trigger needs a shortcut".into());
        }
    }
    Ok(())
}

pub fn defaults(macros: Value) -> Value {
    json!({"format":"yohaku-socius/1","macros":macros,"chat":{"x":null,"y":null,"width":880,"height":720,"font_scale":3}})
}
pub fn validate_settings(v: &Value) -> Result<(), String> {
    let obj = v.as_object().ok_or("Socius settings must be an object")?;
    if obj
        .keys()
        .any(|k| !["format", "macros", "chat", "codex_entry"].contains(&k.as_str()))
    {
        return Err("Unknown Socius setting".into());
    }
    if v["format"] != "yohaku-socius/1" {
        return Err("format must be yohaku-socius/1".into());
    }
    validate_macros(&v["macros"])?;
    let chat = v["chat"].as_object().ok_or("chat must be an object")?;
    if chat
        .keys()
        .any(|k| !["x", "y", "width", "height", "font_scale", "backend", "provider", "limits"].contains(&k.as_str()))
    {
        return Err("Unknown chat setting".into());
    }
    if let Some(backend) = chat.get("backend") {
        if !["agent", "codex-local"].iter().any(|value| backend.as_str() == Some(value)) {
            return Err("chat.backend must be agent or codex-local".into());
        }
    }
    if let Some(limits) = chat.get("limits") {
        let limits=limits.as_object().ok_or("chat.limits must be an object")?;
        for (key,value) in limits {
            match key.as_str() {
                "turn_timeout" if value.as_u64().is_some_and(|n| n>0 && n<=i64::MAX as u64) => (),
                "turn_timeout_long" if value.is_boolean() => (),
                "request_max_retries" if value.as_u64().is_some_and(|n| n<=5) => (),
                _ => return Err(format!("Invalid chat.limits.{key}; positive seconds, boolean long preset, retries 0..5")),
            }
        }
    }
    if let Some(provider) = chat.get("provider") {
        let provider = provider.as_object().ok_or("chat.provider must be an object")?;
        if provider.keys().any(|key| !["protocol","base_url","model","api_key_env","credential_source","max_tokens","timeout","request_options","name","personality"].contains(&key.as_str())) {
            return Err("Unknown/public-only chat.provider field; never store keys here".into());
        }
    }
    if let Some(entry) = v.get("codex_entry") {
        if !entry.as_str().is_some_and(|value| value.starts_with('/') && value.len() <= 4096 && !value.chars().any(char::is_control)) {
            return Err("codex_entry must be an absolute Linux/WSL executable address".into());
        }
    }
    for (key, min, max) in [
        ("width", 480, 4096),
        ("height", 420, 4096),
        ("font_scale", 2, 4),
    ] {
        let n = v["chat"][key]
            .as_i64()
            .ok_or_else(|| format!("chat.{key} must be an integer"))?;
        if !(min..=max).contains(&n) {
            return Err(format!("chat.{key} must be {min}..{max}"));
        }
    }
    for key in ["x", "y"] {
        let val = &v["chat"][key];
        if !val.is_null() && !val.as_i64().is_some_and(|n| (0..=32768).contains(&n)) {
            return Err(format!("chat.{key} must be null or 0..32768"));
        }
    }
    Ok(())
}
pub fn parse_settings(text: &str) -> Result<Value, String> {
    if text.len() > MAX_SETTINGS_BYTES {
        return Err("Socius settings exceed 256 KiB".into());
    }
    let v =
        serde_json::from_str::<Value>(text).map_err(|e| format!("Invalid settings JSON: {e}"))?;
    validate_settings(&v)?;
    Ok(v)
}
pub fn read_settings(root: &std::path::Path) -> Result<Option<String>, String> {
    use std::io::Read;
    let mut path = root.to_owned();
    for part in SETTINGS_PATH.split('/') {
        path.push(part);
        match std::fs::symlink_metadata(&path) {
            Ok(m) if m.file_type().is_symlink() => {
                return Err("Socius settings cannot use symlinks".into());
            }
            Ok(_) => (),
            Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
            Err(e) => return Err(e.to_string()),
        }
    }
    let mut text = String::new();
    std::fs::File::open(path)
        .map_err(|e| e.to_string())?
        .take((MAX_SETTINGS_BYTES + 1) as u64)
        .read_to_string(&mut text)
        .map_err(|e| e.to_string())?;
    if text.len() > MAX_SETTINGS_BYTES {
        return Err("Socius settings exceed 256 KiB".into());
    }
    Ok(Some(text))
}

pub fn entry_config(settings: &Value, legacy: &Value, backend: &str) -> Result<Value, String> {
    validate_settings(settings)?;
    let mut config = legacy.clone();
    let object = config.as_object_mut().ok_or("Invalid host AI configuration")?;
    for key in ["api_key", "credential_source", "protocol", "base_url", "model", "name", "codex_entry"] {
        object.remove(key);
    }
    match backend {
        "agent" => {
            let provider = settings["chat"]["provider"].as_object().ok_or("Missing chat.provider in memory/socius.json")?;
            if !["chat", "responses", "anthropic"].iter().any(|value| provider.get("protocol").and_then(Value::as_str) == Some(value)) {
                return Err("chat.provider.protocol must be chat, responses or anthropic".into());
            }
            for key in ["base_url", "model"] {
                if !provider.get(key).and_then(Value::as_str).is_some_and(|value| !value.trim().is_empty()) {
                    return Err(format!("Missing chat.provider.{key}"));
                }
            }
            config["name"] = json!("Yohaku Agent");
            for (key, value) in provider { config[key] = value.clone(); }
        }
        "codex-local" => {
            let entry = settings["codex_entry"].as_str().filter(|value| !value.trim().is_empty()).ok_or("Missing codex_entry in memory/socius.json")?;
            config["protocol"] = json!("codex-local");
            config["codex_entry"] = json!(entry);
            config["name"] = json!("Codex");
            config["model"] = legacy["model"].clone();
        }
        _ => return Err("chat.backend must be agent or codex-local".into()),
    }
    apply_task_limits(settings, &mut config);
    Ok(config)
}

pub fn apply_task_limits(settings: &Value, config: &mut Value) {
    if let Some(limits)=settings["chat"]["limits"].as_object() {
        for (key,value) in limits { config[key]=value.clone(); }
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn schema_bounds_reject_invalid_values_and_unknown_actions() {
        let valid = defaults(default_macros());
        assert!(validate_settings(&valid).is_ok());
        for (key, value) in [
            ("width", json!(479)),
            ("height", json!(419)),
            ("font_scale", json!(5)),
            ("x", json!(-1)),
            ("y", json!("auto")),
            ("width", json!(true)),
        ] {
            let mut v = valid.clone();
            v["chat"][key] = value;
            assert!(validate_settings(&v).is_err());
        }
        let mut v = valid.clone();
        v["chat"]["command"] = json!("run");
        assert!(validate_settings(&v).is_err());
        let mut v = valid.clone();
        v["macros"][0]["action"] = json!("exec");
        assert!(validate_settings(&v).is_err());
        assert!(parse_settings(&" ".repeat(MAX_SETTINGS_BYTES + 1)).is_err());
    }
    #[test]
    fn task_limits_reach_both_backends_and_invalid_limits_are_rejected() {
        let mut s=defaults(default_macros());
        s["chat"]["provider"]=json!({"protocol":"chat","base_url":"http://127.0.0.1:1","model":"offline"});
        s["codex_entry"]=json!("/offline/codex");
        s["chat"]["limits"]=json!({"turn_timeout":7,"turn_timeout_long":true,"request_max_retries":1});
        for backend in ["agent","codex-local"] {
            let c=entry_config(&s,&json!({"permissions":{"read":true}}),backend).unwrap();
            assert_eq!(c["turn_timeout"],7);assert_eq!(c["turn_timeout_long"],true);
            assert_eq!(c["request_max_retries"],1);assert_eq!(c["permissions"]["read"],true);
        }
        for bad in [json!({"turn_timeout":0}),json!({"turn_timeout":true}),json!({"turn_timeout_long":1}),json!({"request_max_retries":6}),json!({"network_timeout":3})] {
            s["chat"]["limits"]=bad;assert!(validate_settings(&s).is_err());
        }
    }
    #[cfg(unix)]
    #[test]
    fn settings_loader_rejects_symlinks_without_following_them() {
        let base =
            std::env::temp_dir().join(format!("yohaku-setting-symlink-{}", std::process::id()));
        std::fs::create_dir_all(&base).unwrap();
        let target = base.join("outside");
        std::fs::write(&target, "secret").unwrap();
        let root = base.join("world");
        std::fs::create_dir_all(&root).unwrap();
        std::os::unix::fs::symlink(&target, root.join("memory")).unwrap();
        assert!(read_settings(&root).is_err());
        assert_eq!(std::fs::read_to_string(target).unwrap(), "secret");
        std::fs::remove_dir_all(base).unwrap();
    }
}
